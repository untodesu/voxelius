if(BUILD_CLIENT)
    add_executable(vgame "${CMAKE_CURRENT_LIST_DIR}/launch.cc")
    target_compile_definitions(vgame PUBLIC VGAME_CLIENT)
    target_include_directories(vgame PUBLIC ${CMAKE_SOURCE_DIR})
    target_link_libraries(vgame PUBLIC client shared)

    if(WIN32)
        enable_language(RC)
        target_sources(vgame PRIVATE "${CMAKE_CURRENT_LIST_DIR}/resource/vgame.rc")
    endif()
endif()

if(BUILD_SERVER)
    add_executable(vgame-server "${CMAKE_CURRENT_LIST_DIR}/launch.cc")
    target_compile_definitions(vgame-server PUBLIC VGAME_SERVER)
    target_include_directories(vgame-server PUBLIC ${CMAKE_SOURCE_DIR})
    target_link_libraries(vgame-server PUBLIC server shared)

    if(WIN32)
        enable_language(RC)
        target_sources(vgame-server PRIVATE "${CMAKE_CURRENT_LIST_DIR}/resource/vgame-server.rc")
    endif()
endif()
